2) ACCOUNT STATEMENT (PDF)
--------------------------------------------------------------------------------

  GET /api/reports/account-statement

  Query parameters
    customerId  string   preferred   PMS Customers.CustomerID
    accountNo   string   alias       Same as customerId (legacy ASP.NET name)

  At least one of customerId or accountNo is required.

  Examples
    Local:
      GET http://127.0.0.1:8000/api/reports/account-statement?customerId=CK00004

    Production:
      GET http://34.131.132.158:8000/api/reports/account-statement?customerId=CK00004

    Legacy alias:
      GET http://34.131.132.158:8000/api/reports/account-statement?accountNo=CK00004

  Success
    Status       : 200
    Content-Type : application/pdf
    Headers      : Content-Disposition: inline; filename="AccountStatement_{customerId}.pdf"
    Body         : PDF binary

  Not found
    Status       : 404
    Content-Type : application/json

    {
      "success": false,
      "message": "Customer not found: CK00004"
    }

  Error
    Status       : 500
    Content-Type : application/json

    {
      "success": false,
      "message": "Report generation failed"
    }

  Validation error (missing customerId/accountNo)
    Status       : 422
    Content-Type : application/json

    {
      "success": false,
      "message": "customerId is required"
    }

  Business logic
    - Load Customer + Project + PaymentPlan schedules + Payments
    - Attach payments by ScheduleID + CustomerID
    - Compute surcharge per schedule (same rules as ASP.NET SurchargeService)
    - Include other ScheduleID-null surcharge payments
    - Render RDLC-equivalent PDF (logo, header, schedule table, totals, footer)

ASP.NET CORE — HttpClient EXAMPLE
--------------------------------------------------------------------------------

  // appsettings.json
  // Local:
  // "ReportService": { "BaseUrl": "http://127.0.0.1:8000" }
  // Production (same GCP account):
  // "ReportService": { "BaseUrl": "http://34.131.132.158:8000" }

  // Program.cs / DI
  // builder.Services.AddHttpClient("ReportService", c =>
  // {
  //     c.BaseAddress = new Uri(builder.Configuration["ReportService:BaseUrl"]!);
  //     c.Timeout = TimeSpan.FromSeconds(60);
  // });

  public async Task<byte[]> GetAccountStatementPdfAsync(string customerId, CancellationToken ct = default)
  {
      var url = $"/api/reports/account-statement?customerId={Uri.EscapeDataString(customerId)}";

      using var response = await _httpClient.GetAsync(url, ct);

      if (response.StatusCode == System.Net.HttpStatusCode.NotFound)
          throw new KeyNotFoundException($"Customer not found: {customerId}");

      response.EnsureSuccessStatusCode();

      var contentType = response.Content.Headers.ContentType?.MediaType;
      if (!string.Equals(contentType, "application/pdf", StringComparison.OrdinalIgnoreCase))
          throw new InvalidOperationException("Report service did not return a PDF.");

      return await response.Content.ReadAsByteArrayAsync(ct);
  }

  // MVC / API controller usage
  // return File(pdfBytes, "application/pdf", $"AccountStatement_{customerId}.pdf");
  // or inline:
  // return File(pdfBytes, "application/pdf");
